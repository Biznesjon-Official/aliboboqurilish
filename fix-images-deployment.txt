2. Check if images exist in the uploads folder:

   nano /etc/nginx/sites-available/aliboboqurilish.uz

4. Make sure this section exists in your Nginx config:
   location /uploads/ {

       proxy_set_header Host $host;

       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;

       add_header Cache-Control "public, immutable";

       alias /opt/alibobo/build/static/;

   nginx -t
   systemctl reload nginx
